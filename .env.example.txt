# --- API ----------------------------------------------------------------------
API_HOST=0.0.0.0
API_PORT=8000

# --- Logging --------------------------------------------------------------------
# One of: DEBUG | INFO | WARNING | ERROR
LOG_LEVEL=INFO

# --- Ingestion (arXiv) ------------------------------------------------------------
ARXIV_CATEGORY=cs.AI
ARXIV_MAX_RESULTS=10
# arXiv's official rate limit is 3 requests/second; keep this at or above 3.

# Where raw ingested metadata + LaTeX source tarballs are written.
# Matches the volume mounted in docker-compose.yml (added in a later milestone).
RAW_DATA_DIR=data/raw

# --- Parsing -----------------------------------------------------------------
# Where parsed, chunked paper content is written.
PROCESSED_DATA_DIR=data/processed
# Soft max characters per chunk (a chunk only exceeds this if a single
# paragraph, kept intact, is already longer on its own).
PARSING_MAX_CHUNK_CHARS=1200

# --- Embeddings ---------------------------------------------------------------
# Local sentence-transformers model — downloaded once (cached), no API key needed.
EMBEDDING_MODEL=all-MiniLM-L6-v2
EMBEDDING_BATCH_SIZE=32

# --- Vector store ---
# Directory (mounted volume in docker-compose.yml) where the FAISS index and
# metadata sidecar are persisted.
VECTOR_STORE_DIR=data/vector_store
# Number of chunks retrieved per query, and the cap on how many of those may
# come from a single paper 
RETRIEVAL_TOP_K=6
RETRIEVAL_MAX_CHUNKS_PER_PAPER=2

# --- LLM (synthesis) ---------------------------------------------------------
# Which provider to use for answer synthesis. One of: groq | openai
